## Код для создания БД:



```
-- 1. Таблица пользователей (users)
CREATE TABLE IF NOT EXISTS users (
    id BIGINT PRIMARY KEY,                            -- Telegram ID пользователя
    first_name VARCHAR(100),                          -- Имя пользователя
    gender VARCHAR(10),                               -- 'male' / 'female'
    age INT,                                           -- Возраст
    height FLOAT,                                      -- Рост в см
    weight FLOAT,                                      -- Вес в кг
    activity_level FLOAT,                              -- Коэффициент активности
    goal VARCHAR(20),                                  -- 'weight_loss', 'muscle_gain', 'maintain'
    daily_calories INT,                                -- Целевая суточная норма ккал
    timezone VARCHAR(50) DEFAULT 'UTC',                -- Часовой пояс IANA
    day_change_hour INT NOT NULL DEFAULT 4,            -- Час смены суток (по умолчанию 04:00)
    last_active_at BIGINT,                             -- Unix Timestamp последней активности
    created_at BIGINT NOT NULL                         -- Unix Timestamp регистрации
);

-- 2. Таблица записей дневника питания (food_logs)
CREATE TABLE IF NOT EXISTS food_logs (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY, -- Авто-инкрементный ID записи
    user_id BIGINT NOT NULL,                           -- Ссылка на users.id
    title VARCHAR(255) NOT NULL,                        -- Название блюда
    calories INT NOT NULL,                             -- Калории (ккал)
    proteins FLOAT NOT NULL,                           -- Белки (г)
    fats FLOAT NOT NULL,                               -- Жиры (г)
    carbs FLOAT NOT NULL,                              -- Углеводы (г)
    portion_g FLOAT,                                   -- Вес порции (г)
    logged_date VARCHAR(10) NOT NULL,                  -- Дата YYYY-MM-DD с учётом сдвига
    created_at BIGINT NOT NULL,                        -- Unix Timestamp создания
    details_json JSONB,                                -- Сырой JSON ответа Gemini

    CONSTRAINT fk_food_logs_user 
        FOREIGN KEY (user_id) 
        REFERENCES users(id) 
        ON DELETE CASCADE
);

-- Индекс для быстрого получения дневника пользователя за конкретный день
CREATE INDEX IF NOT EXISTS idx_food_logs_user_date 
    ON food_logs (user_id, logged_date);


-- 3. Таблица напоминаний / витаминов (reminders)
CREATE TABLE IF NOT EXISTS reminders (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY, -- Авто-инкрементный ID напоминания
    user_id BIGINT NOT NULL,                           -- Ссылка на users.id
    title VARCHAR(255) NOT NULL,                        -- Текст напоминания
    time_start VARCHAR(5) NOT NULL,                    -- Нач. окна ('12:00')
    time_end VARCHAR(5) NOT NULL,                      -- Кон. окна ('15:00')
    min_calories INT NOT NULL DEFAULT 0,               -- Мин. ккал для срабатывания
    is_triggered_today BOOLEAN NOT NULL DEFAULT FALSE, -- Флаг срабатывания за день
    is_active BOOLEAN NOT NULL DEFAULT TRUE,           -- Статус (вкл/выкл)

    CONSTRAINT fk_reminders_user 
        FOREIGN KEY (user_id) 
        REFERENCES users(id) 
        ON DELETE CASCADE
);

-- Индекс для быстрого поиска активных напоминаний пользователя
CREATE INDEX IF NOT EXISTS idx_reminders_user_active 
    ON reminders (user_id, is_active);
```
